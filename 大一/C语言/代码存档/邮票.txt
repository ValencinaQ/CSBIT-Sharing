#include <stdio.h>
#include <stdlib.h>
#include <math.h>
int main()
{
	int a,b,c,d;
	int m,n,p,q;
	int i=1;
	scanf("%d %d %d %d",&a,&b,&c,&d);
A:for(m=0;m<=5;m++)
    {
    	for(n=0;n<=5-m;n++)
    	{
    		for(p=0;p<=5-m-n;p++)
    		{
    			for(q=0;q<=5-m-n-p;q++)
    			{
    				if(m*a+n*b+p*c+q*d==i)
    				{
    					i++;
    					goto A;
					}
				}
			}
		}
	}
	printf("The max is %d.",i-1);
	return 0;
}
