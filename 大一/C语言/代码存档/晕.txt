#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <math.h>
int main()
{
	int n,i,j,m,k,t,a,b,c,d,e,f,g,h,p;
	scanf("%d",&n);
	int s[30][30]={0};
	for(j=0;j<n;j++)
	{
		s[0][j]=j+1;
	}
	t=j;
	for(i=0;i<n;i++)
	{
		s[i][j-1]=t;
		t++;
	}
	t=t-1;
	for(j=n-1;j>=0;j--)
	{
		s[i-1][j]=t;
		t++;
	}
	t=t-1;
	for(i=n-1;i>0;i--)
	{
		s[i][j+1]=t;
		t++;
    }
	a=0;b=n-1;c=1;d=n-1;e=n-2;f=0;g=n-2;h=1;
	for(p=1;p<10;p++) 
	{
		t=t-1;
	    for(j=a;j<b;j++)
	    {
		    s[i+1][j]=t;
		    t++;
	    }		
	    t=t-1;
	    for(i=c;i<d;i++)
	    {
		    s[i][j-1]=t;
		    t++;
	    }		
        t=t-1;	
	    for(j=e;j>f;j--)
	    {
		    s[i-1][j]=t;
		    t++;
	    }	
	    t=t-1;	
	    for(i=g;i>h;i--)
	    {
		    s[i][j+1]=t;
		    t++;
	    }
	    a++;b--;c++;d--;e--;f++;g--;h++;
	}
		for(m=0;m<n;m++)
	{
		for(k=0;k<n;k++)
		{
			printf("%3d",s[m][k]);	
		}
		printf("\n");
	}	
}
